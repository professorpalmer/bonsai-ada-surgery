#!/usr/bin/env bash
#
# rotate_logs.sh
#
# Rotates log files in /var/log using size and age limits.
#
# Usage:
#   ./rotate_logs.sh [LOG_DIR] [SIZE_LIMIT_KB] [AGE_DAYS]
#
# Arguments:
#   LOG_DIR        Directory containing logs (default: /var/log)
#   SIZE_LIMIT_KB  Rotate logs larger than this size in KB (default: 1024)
#   AGE_DAYS       Rotate logs older than this many days (default: 7)
#
# Behavior:
#   - Only processes regular files (skips directories, symlinks, etc.)
#   - Skips files already ending in .gz (prevents double-compression)
#   - Creates timestamped rotated files: <name>.<timestamp>
#   - Compresses rotated files with gzip
#   - Retains only the 5 most recent rotated files per log name
#   - Logs actions to /var/log/rotate_logs.log
#
set -euo pipefail

# --- Configuration (overridable via CLI args) ---
LOG_DIR="${1:-/var/log}"
SIZE_LIMIT_KB="${2:-1024}"
AGE_DAYS="${3:-7}"
MAX_ROTATED=5
LOG_FILE="/var/log/rotate_logs.log"

# --- Helper: log a message ---
log() {
    local msg="[$(date '+%Y-%m-%d %H:%M:%S')] $*"
    echo "$msg" | tee -a "$LOG_FILE"
}

    log "ERROR: Directory '$LOG_DIR' does not exist."
