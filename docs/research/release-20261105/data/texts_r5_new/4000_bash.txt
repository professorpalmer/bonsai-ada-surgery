#!/usr/bin/env bash
#
# rotate_logs.sh
#
# Rotates log files in /var/log using size and age limits.
#
# Usage:
#   ./rotate_logs.sh [LOG_DIR] [SIZE_LIMIT_KB] [AGE_DAYS]
#
# Arguments:
#   LOG_DIR       Directory containing logs (default: /var/log)
#   SIZE_LIMIT_KB Maximum size in KB before rotation (default: 1024 = 1 MB)
#   AGE_DAYS      Maximum age in days before rotation (default: 7)
#
# Behavior:
#   - Only rotates regular files (skips directories, symlinks, sockets, etc.)
#   - Skips files already ending in .gz (prevents double-compression)
#   - Skips files that are currently being written to (open by a process)
#   - Creates timestamped rotated files: <name>.<timestamp>
#   - Compresses rotated files with gzip
#   - Retains only the most recent 5 rotated files per log name
#   - Logs actions to /var/log/rotate_logs.log
#

SIZE_LIMIT_KB="${2:-1024}"
AGE_DAYS="${3:-7}"

LOG_FILE="/var/log/rotate_logs.log"

    echo "ERROR: Log directory '$LOG_DIR' does not exist." >&2
    exit 1
fi
